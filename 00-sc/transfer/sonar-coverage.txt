import unittest
from unittest.mock import MagicMock, patch
from adls_connector import ADLSConnector  # 假设你的文件名是 adls_connector.py

class TestADLSConnector(unittest.TestCase):

    def setUp(self):
        # 准备一个标准的测试 URL
        self.test_url = "abfss://mycontainer@mystorage.dfs.core.windows.net/test/a.txt"

        # 模拟环境变量，防止 WorkloadIdentityCredential 报错
        self.env_patcher = patch.dict('os.environ', {
            "AZURE_CLIENT_ID": "fake-client-id",
            "AZURE_TENANT_ID": "fake-tenant-id",
            "AZURE_FEDERATED_TOKEN_FILE": "/fake/path/token"
        })
        self.env_patcher.start()

    def tearDown(self):
        self.env_patcher.stop()

    def test_parse_abfss_valid(self):
        """测试正常路径解析"""
        connector = ADLSConnector(self.test_url)
        self.assertEqual(connector.container, "mycontainer")
        self.assertEqual(connector.account_name, "mystorage")
        self.assertEqual(connector.file_path, "test/a.txt") # 验证 lstrip 执行成功

    def test_parse_abfss_invalid_protocol(self):
        """测试错误协议校验"""
        with self.assertRaises(ValueError) as cm:
            ADLSConnector("https://storage.com/file")
        self.assertIn("Invalid protocol", str(cm.exception))

    def test_parse_abfss_missing_at(self):
        """测试缺失 @ 符号的校验"""
        with self.assertRaises(ValueError) as cm:
            ADLSConnector("abfss://mystorage.dfs.core.windows.net/test/a.txt")
        self.assertIn("Missing '@'", str(cm.exception))

    @patch('adls_connector.DataLakeServiceClient')
    @patch('adls_connector.WorkloadIdentityCredential')
    def test_read_content_success(self, mock_cred, mock_service_client):
        """测试读取内容成功场景"""
        # 设置 Mock 层级结构: ServiceClient -> FileSystemClient -> FileClient -> Download
        mock_fs_client = MagicMock()
        mock_file_client = MagicMock()
        mock_download = MagicMock()

        mock_service_client.return_value.get_file_system_client.return_value = mock_fs_client
        mock_fs_client.get_file_client.return_value = mock_file_client
        mock_file_client.download_file.return_value = mock_download
        mock_download.readall.return_value = b"Hello Sonar" # 模拟二进制返回

        connector = ADLSConnector(self.test_url)
        content = connector.read_content()

        self.assertEqual(content, "Hello Sonar")
        # 验证解析后的相对路径是否正确传递给了 SDK
        mock_fs_client.get_file_client.assert_called_with("test/a.txt")

    @patch('adls_connector.DataLakeServiceClient')
    def test_read_content_azure_error(self, mock_service_client):
        """测试 Azure SDK 抛出异常时的处理"""
        from azure.core.exceptions import AzureError

        mock_service_client.return_value.get_file_system_client.side_effect = AzureError("Access Denied")

        connector = ADLSConnector(self.test_url)
        result = connector.read_content()

        self.assertIn("Azure Storage Error", result)

if __name__ == '__main__':
    unittest.main()
