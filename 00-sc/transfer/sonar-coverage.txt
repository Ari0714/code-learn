import unittest
from unittest.mock import MagicMock, patch
import os

# Import the class from your target file
from adls_connector import ADLSConnector

class TestADLSConnector(unittest.TestCase):

    def setUp(self):
        """
        Set up the test environment before each test case.
        """
        # CRITICAL: Reset the Singleton instance to ensure __init__ runs for every test.
        # This is essential for achieving high Sonar coverage.
        ADLSConnector._instance = None

        # Initialize the Mock SparkSession
        self.mock_spark = MagicMock()
        self.mock_hadoop_conf = MagicMock()

        # Mock the deep chain: spark.sparkContext._jsc.hadoopConfiguration()
        self.mock_spark.sparkContext._jsc.hadoopConfiguration.return_value = self.mock_hadoop_conf

        # Define the behavior for hadoop_conf.get(key, default)
        def mock_get_config(key, default=None):
            # Simulated Hadoop configuration values
            configs = {
                "fs.azure.account": "teststorageaccount",
                "fs.azure.container": "testcontainer"
            }
            return configs.get(key, default)

        self.mock_hadoop_conf.get.side_effect = mock_get_config

        # Mock necessary environment variables for Workload Identity authentication
        self.env_patcher = patch.dict('os.environ', {
            "AZURE_FEDERATED_TOKEN_FILE": "/fake/token/path",
            "AZURE_CLIENT_ID": "fake-client-id",
            "AZURE_TENANT_ID": "fake-tenant-id"
        })
        self.env_patcher.start()

    def tearDown(self):
        """
        Clean up the environment after each test case.
        """
        self.env_patcher.stop()
        # Ensure the Singleton is reset for other test files
        ADLSConnector._instance = None

    @patch('adls_connector.WorkloadIdentityCredential')
    @patch('adls_connector.DataLakeServiceClient')
    def test_init_success(self, mock_service_client, mock_credential):
        """
        Test successful initialization.
        Passing self.mock_spark avoids the 'str' object has no attribute 'sparkContext' error.
        """
        # Act: Instantiate the connector with a mocked Spark object
        connector = ADLSConnector(self.mock_spark)

        # Assert: Verify that attributes are correctly parsed from Hadoop configuration
        self.assertEqual(connector.az_container, "testcontainer")
        self.assertIn("teststorageaccount", connector.account_url)

        # Assert: Check if Azure clients were instantiated
        mock_credential.assert_called_once()
        mock_service_client.assert_called_once()

    def test_singleton_logic(self):
        """
        Verify that ADLSConnector adheres to the Singleton pattern.
        """
        # Both calls should return the exact same object in memory
        instance1 = ADLSConnector(self.mock_spark)
        instance2 = ADLSConnector(self.mock_spark)

        self.assertIs(instance1, instance2, "Multiple calls must return the same instance.")

    def test_missing_environment_variable(self):
        """
        Verify error handling when the required Azure token file environment variable is missing.
        """
        # Clear environment variables to trigger the 'if not os.environ.get' block in your code
        with patch.dict('os.environ', {}, clear=True):
            ADLSConnector._instance = None
            # Check if the code raises an Exception as expected in your _authenticate method
            with self.assertRaises(Exception):
                ADLSConnector(self.mock_spark)

if __name__ == '__main__':
    unittest.main()
