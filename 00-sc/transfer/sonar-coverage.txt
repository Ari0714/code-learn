import unittest
from unittest.mock import MagicMock, patch
import os

# 假设你的类定义在 adls_connector.py 中
from adls_connector import ADLSConnector

class TestADLSConnector(unittest.TestCase):

    def setUp(self):
        # 1. 彻底重置单例状态（因为你的类使用了 _instance，测试间会互相干扰）
        ADLSConnector._instance = None

        # 2. 构造一个复杂的 Mock Spark 对象
        self.mock_spark = MagicMock()
        self.mock_hadoop_conf = MagicMock()

        # 模拟 spark.sparkContext._jsc.hadoopConfiguration()
        self.mock_spark.sparkContext._jsc.hadoopConfiguration.return_value = self.mock_hadoop_conf

        # 模拟 Hadoop 配置返回值
        self.mock_hadoop_conf.get.side_effect = lambda key, default: {
            "fs.azure.account": "mystorage",
            "fs.azure.container": "mycontainer"
        }.get(key, default)

        # 3. 模拟环境变量防止认证报错
        self.env_patcher = patch.dict('os.environ', {
            "AZURE_FEDERATED_TOKEN_FILE": "/fake/path/token",
            "AZURE_CLIENT_ID": "fake-id",
            "AZURE_TENANT_ID": "fake-tenant"
        })
        self.env_patcher.start()

    def tearDown(self):
        self.env_patcher.stop()
        ADLSConnector._instance = None

    @patch('adls_connector.WorkloadIdentityCredential')
    @patch('adls_connector.DataLakeServiceClient')
    def test_init_success(self, mock_client, mock_cred):
        """测试正常初始化，解决 'str' object 报错"""
        # 这里的关键是：必须传入 mock_spark 对象，而不是字符串路径
        connector = ADLSConnector(self.mock_spark)

        # 验证属性是否正确从 mock_hadoop_conf 提取
        self.assertEqual(connector.az_container, "mycontainer")
        self.assertEqual(connector.account_url, "https://mystorage.dfs.core.windows.net")

        # 验证认证是否被触发
        mock_cred.assert_called_once()

    def test_singleton_behavior(self):
        """测试单例模式覆盖率"""
        connector1 = ADLSConnector(self.mock_spark)
        connector2 = ADLSConnector(self.mock_spark)
        self.assertIs(connector1, connector2)

    @patch('adls_connector.DataLakeServiceClient')
    def test_authenticate_error(self, mock_client):
        """覆盖异常处理逻辑"""
        # 模拟环境变量缺失的情况
        with patch.dict('os.environ', {}, clear=True):
            # 重新初始化会导致触发 _authenticate
            ADLSConnector._instance = None
            with self.assertRaises(Exception): # 或者你定义的特定异常
                ADLSConnector(self.mock_spark)

if __name__ == '__main__':
    unittest.main()
